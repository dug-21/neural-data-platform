# ALGO-state-transition-extract: LAG-Based State Transition Extraction

> **Algorithm ID:** C02
> **Feature:** v11-006 (State Transition Materializer)
> **Phase:** C (Cross-Stream + Alignment)
> **Created:** 2026-02-04

---

## Purpose

Extract discrete state transition events from continuous state event data using LAG window functions. This algorithm transforms raw state events into transition records with `is_actual_transition` filtering, duration calculation, and direction derivation for use in V1.2 pattern detection.

---

## Algorithm: GenerateStateTransitionView

```
ALGORITHM: GenerateStateTransitionView
INPUT:
    stream_config: StreamConfig
    action: Action
OUTPUT: Result<String, GeneratorError>
REQUIRES:
    - stream_config.stream_type == "state_event"
    - stream_config.gold_etl.transitions.enabled = true

BEGIN
    // 1. Validate stream type
    IF stream_config.stream_type != "state_event" THEN
        RETURN Err(GeneratorError::Code401_InvalidStreamType {
            stream_id: stream_config.stream_id,
            expected: "state_event",
            actual: stream_config.stream_type
        })
    END IF

    // 2. Extract transitions config
    transitions <- stream_config.gold_etl.transitions
    IF transitions IS None OR NOT transitions.enabled THEN
        RETURN Err(GeneratorError::TransitionsNotEnabled {
            stream_id: stream_config.stream_id
        })
    END IF

    // 3. Resolve source table
    source_table <- ResolveSilverSourceTable(stream_config)

    // 4. Extract configuration
    state_field <- transitions.state_field         // e.g., "state"
    entity_field <- transitions.entity_field       // e.g., "ndp_id"
    timestamp_field <- "event_time"                // Standard Silver column
    direction_mapping <- transitions.direction_mapping.unwrap_or_default()

    // 5. Generate view name
    view_name <- format!("gold.{}_transitions", stream_config.stream_id.replace("-", "_"))

    // 6. Generate SQL
    sql <- GenerateTransitionSQL(
        view_name,
        source_table,
        state_field,
        entity_field,
        timestamp_field,
        direction_mapping,
        stream_config.stream_id,
        action
    )

    RETURN Ok(sql)
END
```

## Algorithm: GenerateTransitionSQL

```
ALGORITHM: GenerateTransitionSQL
INPUT:
    view_name: String
    source_table: String
    state_field: String
    entity_field: String
    timestamp_field: String
    direction_mapping: Map<String, String>
    stream_id: String
    action: Action
OUTPUT: String

BEGIN
    // Build the direction CASE expression
    direction_expr <- GenerateDirectionExpression(direction_mapping)

    // Build device type expression (derived from entity_id pattern)
    device_type_expr <- GenerateDeviceTypeExpression(entity_field)

    sql <- format!(r#"
-- State Transition View: {view_name}
-- Stream: {stream_id}
-- Generated by ndp-gold-ddl
-- Mode: {action_mode}

{drop_if_recreate}

CREATE MATERIALIZED VIEW {view_name} AS
SELECT
    -- Event identification
    {timestamp_field} AS transition_time,
    {entity_field} AS entity_id,
    '{stream_id}'::TEXT AS stream_id,

    -- State transition details
    LAG({state_field}) OVER w AS from_state,
    {state_field} AS to_state,

    -- Is this an actual state change?
    CASE
        WHEN LAG({state_field}) OVER w IS DISTINCT FROM {state_field} THEN TRUE
        WHEN LAG({state_field}) OVER w IS NULL THEN TRUE  -- First event is a transition
        ELSE FALSE
    END AS is_actual_transition,

    -- Duration in previous state
    {timestamp_field} - LAG({timestamp_field}) OVER w AS duration_in_previous_state,
    EXTRACT(EPOCH FROM ({timestamp_field} - LAG({timestamp_field}) OVER w)) * 1000 AS duration_ms,

    -- Transition direction
    {direction_expr} AS transition_direction,

    -- Device type derived from entity_id
    {device_type_expr} AS device_type

FROM {source_table}
WINDOW w AS (PARTITION BY {entity_field} ORDER BY {timestamp_field});

-- Index for time-based queries
CREATE INDEX IF NOT EXISTS idx_{view_short}_time
    ON {view_name} (transition_time DESC);

-- Index for entity-based queries
CREATE INDEX IF NOT EXISTS idx_{view_short}_entity
    ON {view_name} (entity_id, transition_time DESC);

-- Filtered view for only actual transitions
CREATE OR REPLACE VIEW gold.{stream_short}_transitions_actual AS
SELECT * FROM {view_name}
WHERE is_actual_transition = TRUE;

COMMENT ON MATERIALIZED VIEW {view_name} IS
    'State transitions extracted from {stream_id}. Refresh with Gold layer.';
"#,
        view_name = view_name,
        stream_id = stream_id,
        action_mode = action.to_string(),
        drop_if_recreate = IF action == Recreate THEN
            format!("DROP MATERIALIZED VIEW IF EXISTS {} CASCADE;", view_name)
        ELSE
            ""
        END IF,
        timestamp_field = timestamp_field,
        entity_field = entity_field,
        state_field = state_field,
        direction_expr = direction_expr,
        device_type_expr = device_type_expr,
        source_table = source_table,
        view_short = view_name.split(".").last().replace("_", ""),
        stream_short = stream_id.replace("-", "_")
    )

    RETURN sql
END
```

## Algorithm: GenerateDirectionExpression

```
ALGORITHM: GenerateDirectionExpression
INPUT: direction_mapping: Map<String, String>
OUTPUT: String

BEGIN
    // Default mapping if not provided
    default_mapping <- {
        "off_to_on": "opening",
        "on_to_off": "closing"
    }

    mapping <- IF direction_mapping.is_empty() THEN default_mapping ELSE direction_mapping

    // Build CASE expression
    cases <- Vec::new()

    FOR EACH (transition, direction) IN mapping DO
        // Parse transition pattern "from_to_to"
        parts <- transition.split("_to_")
        IF parts.len() == 2 THEN
            from_state <- parts[0]
            to_state <- parts[1]
            cases.push(format!(
                "WHEN LAG(state) OVER w = '{}' AND state = '{}' THEN '{}'",
                from_state, to_state, direction
            ))
        END IF
    END FOR

    // Add initial state case
    cases.push("WHEN LAG(state) OVER w IS NULL THEN 'initial'")

    // Add default case
    cases.push("ELSE 'unknown'")

    RETURN format!("CASE\n        {}\n    END", cases.join("\n        "))
END
```

## Algorithm: GenerateDeviceTypeExpression

```
ALGORITHM: GenerateDeviceTypeExpression
INPUT: entity_field: String
OUTPUT: String

BEGIN
    // Derive device type from entity_id naming pattern
    // Example: door_backslider -> door, window_office -> window
    RETURN format!(r#"
CASE
        WHEN {} LIKE 'door_%' THEN 'door'
        WHEN {} LIKE 'window_%' THEN 'window'
        WHEN {} LIKE 'hvac_%' THEN 'hvac'
        WHEN {} LIKE 'light_%' THEN 'light'
        ELSE 'other'
    END"#,
        entity_field, entity_field, entity_field, entity_field
    )
END
```

## Algorithm: DetectTransitions (Runtime Query Logic)

```
ALGORITHM: DetectTransitions
INPUT:
    events: Vec<StateEvent>     // Sorted by entity_id, event_time
OUTPUT: Vec<Transition>

DESCRIPTION:
    This describes the runtime behavior when the view is queried.
    The SQL implementation uses window functions.

BEGIN
    transitions <- Vec::new()

    // Group by entity
    events_by_entity <- events.group_by(|e| e.entity_id)

    FOR EACH (entity_id, entity_events) IN events_by_entity DO
        prev_state <- None
        prev_time <- None

        FOR EACH event IN entity_events DO
            // Calculate is_actual_transition
            is_actual <- MATCH prev_state WITH
                | None => TRUE                              // First event
                | Some(ps) => ps IS DISTINCT FROM event.state  // State changed
            END

            // Calculate duration
            duration <- MATCH prev_time WITH
                | None => None
                | Some(pt) => Some(event.event_time - pt)
            END

            // Calculate direction
            direction <- DeriveDirection(prev_state, event.state)

            // Build transition record
            transition <- Transition {
                transition_time: event.event_time,
                entity_id: entity_id,
                stream_id: event.stream_id,
                from_state: prev_state,
                to_state: event.state,
                is_actual_transition: is_actual,
                duration_in_previous_state: duration,
                transition_direction: direction
            }

            transitions.push(transition)

            // Update state for next iteration
            prev_state <- Some(event.state)
            prev_time <- Some(event.event_time)
        END FOR
    END FOR

    RETURN transitions
END
```

## Algorithm: DeriveDirection

```
ALGORITHM: DeriveDirection
INPUT:
    from_state: Option<String>
    to_state: String
OUTPUT: TransitionDirection

BEGIN
    RETURN MATCH (from_state, to_state) WITH
        | (None, _) => TransitionDirection::Initial
        | (Some("off"), "on") => TransitionDirection::Opening
        | (Some("on"), "off") => TransitionDirection::Closing
        | _ => TransitionDirection::Unknown
END
```

## SQL Example: State Transitions View

```sql
-- Generated state transitions view
CREATE MATERIALIZED VIEW gold.state_events_transitions AS
SELECT
    event_time AS transition_time,
    ndp_id AS entity_id,
    'home-assistant-state'::TEXT AS stream_id,

    LAG(state) OVER w AS from_state,
    state AS to_state,

    CASE
        WHEN LAG(state) OVER w IS DISTINCT FROM state THEN TRUE
        WHEN LAG(state) OVER w IS NULL THEN TRUE
        ELSE FALSE
    END AS is_actual_transition,

    event_time - LAG(event_time) OVER w AS duration_in_previous_state,
    EXTRACT(EPOCH FROM (event_time - LAG(event_time) OVER w)) * 1000 AS duration_ms,

    CASE
        WHEN LAG(state) OVER w = 'off' AND state = 'on' THEN 'opening'
        WHEN LAG(state) OVER w = 'on' AND state = 'off' THEN 'closing'
        WHEN LAG(state) OVER w IS NULL THEN 'initial'
        ELSE 'unknown'
    END AS transition_direction,

    CASE
        WHEN ndp_id LIKE 'door_%' THEN 'door'
        WHEN ndp_id LIKE 'window_%' THEN 'window'
        ELSE 'other'
    END AS device_type

FROM silver.state_events
WINDOW w AS (PARTITION BY ndp_id ORDER BY event_time);

CREATE INDEX idx_state_events_transitions_time
    ON gold.state_events_transitions (transition_time DESC);

CREATE INDEX idx_state_events_transitions_entity
    ON gold.state_events_transitions (entity_id, transition_time DESC);

-- Filtered view for actual transitions only
CREATE OR REPLACE VIEW gold.state_events_transitions_actual AS
SELECT * FROM gold.state_events_transitions
WHERE is_actual_transition = TRUE;
```

## Query Examples

### Get Actual Transitions for Last 24 Hours

```sql
SELECT
    transition_time,
    entity_id,
    from_state,
    to_state,
    duration_in_previous_state,
    transition_direction,
    device_type
FROM gold.state_events_transitions_actual
WHERE transition_time >= NOW() - INTERVAL '24 hours'
ORDER BY transition_time DESC;
```

### Count Transitions by Device Type

```sql
SELECT
    device_type,
    transition_direction,
    COUNT(*) as count
FROM gold.state_events_transitions_actual
WHERE transition_time >= NOW() - INTERVAL '7 days'
GROUP BY device_type, transition_direction
ORDER BY device_type, count DESC;
```

### Average Duration in State by Entity

```sql
SELECT
    entity_id,
    to_state,
    AVG(duration_ms) / 1000 / 60 as avg_duration_minutes
FROM gold.state_events_transitions_actual
WHERE duration_ms IS NOT NULL
GROUP BY entity_id, to_state
ORDER BY entity_id, avg_duration_minutes DESC;
```

## Complexity Analysis

| Operation | Time Complexity | Space Complexity |
|-----------|-----------------|------------------|
| Window function LAG | O(n log n) | O(n) |
| is_actual_transition | O(1) per row | O(1) |
| Duration calculation | O(1) per row | O(1) |
| Direction derivation | O(1) per row | O(1) |
| Total view creation | O(n log n) | O(n) |

Where n = number of state events in source table

## Invariants

1. **State Event Only**: Only `state_event` streams can have transitions
2. **Partitioned by Entity**: Window function partitions by entity_id
3. **Ordered by Time**: Events ordered by timestamp within partition
4. **First Event is Transition**: NULL from_state counts as transition
5. **Duplicate Filtering**: is_actual_transition = FALSE for same-state events
6. **Duration Requires Previous**: duration_ms is NULL for first event
