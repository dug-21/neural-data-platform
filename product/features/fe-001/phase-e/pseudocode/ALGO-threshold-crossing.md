# ALGO-threshold-crossing: Crossing Detection with LAG Comparison

> **Algorithm ID:** E01
> **Feature:** v11-012 (Threshold Crossing Generator)
> **Phase:** E (Unified Event Abstraction)
> **Created:** 2026-02-04

---

## Purpose

Detect when observation metrics cross objective thresholds by comparing consecutive hourly readings using LAG window functions. This algorithm generates crossing events that feed into the unified events view for V1.2 pattern detection, enabling correlation between state changes and metric violations.

---

## Algorithm: GenerateThresholdCrossingView

```
ALGORITHM: GenerateThresholdCrossingView
INPUT:
    domain_config: DomainConfig
    config_loader: ConfigLoader
OUTPUT: Result<String, GeneratorError>
REQUIRES:
    - domain_config.objectives is non-empty
    - All objective target streams have Gold layer enabled

BEGIN
    domain_id <- domain_config.id
    objectives <- domain_config.objectives

    // 1. Validate objectives
    validation_result <- ValidateObjectives(objectives, config_loader)
    IF validation_result.has_errors() THEN
        RETURN Err(GeneratorError::InvalidObjectives {
            errors: validation_result.errors
        })
    END IF

    // 2. Group objectives by stream
    objectives_by_stream <- objectives.group_by(|o| o.target.stream)

    // 3. Generate CTEs for each stream
    stream_ctes <- Vec::new()
    FOR EACH (stream_id, stream_objectives) IN objectives_by_stream DO
        cte <- GenerateStreamCTE(stream_id, stream_objectives, domain_id)
        stream_ctes.push(cte)
    END FOR

    // 4. Generate objectives CTE
    objectives_cte <- GenerateObjectivesCTE(objectives)

    // 5. Generate crossing detection SQL
    crossing_sql <- GenerateCrossingSQL(
        domain_id,
        objectives_cte,
        stream_ctes
    )

    // 6. Wrap in view creation
    view_name <- format!("gold.{}_threshold_crossings", domain_id.replace("-", "_"))

    sql <- format!(r#"
-- Threshold Crossing View for domain: {domain_id}
-- Generated by ndp-gold-ddl
-- Detects when metrics cross objective thresholds

CREATE OR REPLACE VIEW {view_name} AS

{crossing_sql}
"#,
        domain_id = domain_id,
        view_name = view_name,
        crossing_sql = crossing_sql
    )

    RETURN Ok(sql)
END
```

## Algorithm: GenerateObjectivesCTE

```
ALGORITHM: GenerateObjectivesCTE
INPUT: objectives: Vec<Objective>
OUTPUT: String

BEGIN
    objective_rows <- Vec::new()

    FOR EACH objective IN objectives DO
        // Handle threshold based on condition type
        threshold_sql <- MATCH objective.target.threshold WITH
            | Threshold::Single(value) =>
                format!("{}", value)
            | Threshold::Range(min, max) =>
                format!("ARRAY[{}, {}]", min, max)
        END

        // Build time window SQL (if applicable)
        time_window_sql <- MATCH objective.time_window WITH
            | Some(window) =>
                format!("'{}', '{}'", window.start, window.end)
            | None =>
                "NULL, NULL"
        END

        row <- format!(r#"
    SELECT
        '{objective_id}'::TEXT AS objective_id,
        '{stream_id}'::TEXT AS stream_id,
        '{metric}'::TEXT AS metric,
        '{condition}'::TEXT AS condition,
        {threshold}::NUMERIC AS threshold,
        {threshold_min}::NUMERIC AS threshold_min,
        {threshold_max}::NUMERIC AS threshold_max,
        '{unit}'::TEXT AS unit,
        {time_window_start}::TIME AS time_window_start,
        {time_window_end}::TIME AS time_window_end"#,
            objective_id = objective.id,
            stream_id = objective.target.stream,
            metric = objective.target.metric,
            condition = objective.target.condition.to_sql_string(),
            threshold = IF objective.target.threshold.is_single() THEN threshold_sql ELSE "NULL",
            threshold_min = IF objective.target.threshold.is_range() THEN
                objective.target.threshold.min().to_string()
            ELSE "NULL",
            threshold_max = IF objective.target.threshold.is_range() THEN
                objective.target.threshold.max().to_string()
            ELSE "NULL",
            unit = objective.target.unit.unwrap_or(""),
            time_window_start = IF objective.time_window.is_some() THEN
                format!("'{}'", objective.time_window.unwrap().start)
            ELSE "NULL",
            time_window_end = IF objective.time_window.is_some() THEN
                format!("'{}'", objective.time_window.unwrap().end)
            ELSE "NULL"
        )

        objective_rows.push(row)
    END FOR

    RETURN format!(
        "objective_thresholds AS (\n{}\n)",
        objective_rows.join("\n    UNION ALL\n")
    )
END
```

## Algorithm: GenerateStreamCTE

```
ALGORITHM: GenerateStreamCTE
INPUT:
    stream_id: String
    objectives: Vec<Objective>
    domain_id: String
OUTPUT: String

BEGIN
    // Get unique metrics from objectives
    metrics <- objectives.iter()
        .map(|o| o.target.metric)
        .collect::<HashSet<_>>()

    gold_table <- format!("gold.{}_hourly", stream_id.replace("-", "_"))

    // Build LAG expressions for each metric
    lag_columns <- Vec::new()
    FOR EACH metric IN metrics DO
        metric_col <- format!("{}_mean", metric)
        lag_columns.push(format!(
            "{} AS {}_value,\n        LAG({}) OVER (PARTITION BY ndp_id ORDER BY bucket) AS {}_prev",
            metric_col, metric, metric_col, metric
        ))
    END FOR

    cte_name <- format!("{}_observations", stream_id.replace("-", "_"))

    RETURN format!(r#"
{cte_name} AS (
    SELECT
        bucket,
        ndp_id AS entity_id,
        '{stream_id}'::TEXT AS stream_id,
        {lag_columns}
    FROM {gold_table}
)"#,
        cte_name = cte_name,
        stream_id = stream_id,
        lag_columns = lag_columns.join(",\n        "),
        gold_table = gold_table
    )
END
```

## Algorithm: GenerateCrossingSQL

```
ALGORITHM: GenerateCrossingSQL
INPUT:
    domain_id: String
    objectives_cte: String
    stream_ctes: Vec<String>
OUTPUT: String

BEGIN
    // Build unpivot CTEs for each stream
    unpivot_ctes <- Vec::new()
    FOR EACH stream_cte IN stream_ctes DO
        stream_name <- ExtractCTEName(stream_cte)
        unpivot <- GenerateUnpivotCTE(stream_name)
        unpivot_ctes.push(unpivot)
    END FOR

    // Build crossing detection
    crossing_select <- GenerateCrossingSelect()

    sql <- format!(r#"
WITH
-- Objective thresholds (inlined from domain config)
{objectives_cte},

-- Per-stream observations with LAG
{stream_ctes},

-- Unpivot to metric rows
observation_metrics AS (
{unpivot_union}
),

-- Detect crossings
crossings AS (
{crossing_select}
)

-- Final output with event schema
SELECT
    gen_random_uuid() AS event_id,
    event_time,
    stream_id,
    entity_id,
    'threshold_crossing'::TEXT AS event_type,
    jsonb_build_object(
        'metric', metric,
        'threshold', threshold,
        'direction', direction,
        'value', value,
        'previous_value', prev_value,
        'objective_id', objective_id,
        'condition', condition,
        'unit', unit
    ) AS details
FROM crossings
WHERE direction IS NOT NULL
"#,
        objectives_cte = objectives_cte,
        stream_ctes = stream_ctes.join(",\n\n"),
        unpivot_union = unpivot_ctes.join("\n    UNION ALL\n"),
        crossing_select = crossing_select
    )

    RETURN sql
END
```

## Algorithm: GenerateCrossingSelect

```
ALGORITHM: GenerateCrossingSelect
OUTPUT: String

BEGIN
    RETURN r#"
SELECT
    om.bucket AS event_time,
    om.stream_id,
    om.entity_id,
    ot.objective_id,
    ot.metric,
    COALESCE(ot.threshold, ot.threshold_min) AS threshold,
    ot.condition,
    ot.unit,
    om.value,
    om.prev_value,
    CASE
        -- Less than condition: crossing is when value >= threshold
        WHEN ot.condition = '<' THEN
            CASE
                WHEN om.prev_value < ot.threshold AND om.value >= ot.threshold THEN 'rising'
                WHEN om.prev_value >= ot.threshold AND om.value < ot.threshold THEN 'falling'
            END

        -- Less than or equal: crossing is when value > threshold
        WHEN ot.condition = '<=' THEN
            CASE
                WHEN om.prev_value <= ot.threshold AND om.value > ot.threshold THEN 'rising'
                WHEN om.prev_value > ot.threshold AND om.value <= ot.threshold THEN 'falling'
            END

        -- Greater than: crossing is when value <= threshold
        WHEN ot.condition = '>' THEN
            CASE
                WHEN om.prev_value > ot.threshold AND om.value <= ot.threshold THEN 'rising'
                WHEN om.prev_value <= ot.threshold AND om.value > ot.threshold THEN 'falling'
            END

        -- Greater than or equal: crossing is when value < threshold
        WHEN ot.condition = '>=' THEN
            CASE
                WHEN om.prev_value >= ot.threshold AND om.value < ot.threshold THEN 'rising'
                WHEN om.prev_value < ot.threshold AND om.value >= ot.threshold THEN 'falling'
            END

        -- Between condition: detect entering/exiting range
        WHEN ot.condition = 'between' THEN
            CASE
                -- Was outside, now inside
                WHEN (om.prev_value < ot.threshold_min OR om.prev_value > ot.threshold_max)
                     AND (om.value >= ot.threshold_min AND om.value <= ot.threshold_max)
                    THEN 'entering_range'
                -- Was inside, now below
                WHEN (om.prev_value >= ot.threshold_min AND om.prev_value <= ot.threshold_max)
                     AND om.value < ot.threshold_min
                    THEN 'exiting_range_low'
                -- Was inside, now above
                WHEN (om.prev_value >= ot.threshold_min AND om.prev_value <= ot.threshold_max)
                     AND om.value > ot.threshold_max
                    THEN 'exiting_range_high'
            END
    END AS direction
FROM observation_metrics om
JOIN objective_thresholds ot
    ON om.stream_id = ot.stream_id
    AND om.metric = ot.metric
WHERE
    -- Both values must be present
    om.value IS NOT NULL
    AND om.prev_value IS NOT NULL
    -- Apply time window if specified
    AND (ot.time_window_start IS NULL
         OR om.bucket::TIME BETWEEN ot.time_window_start AND ot.time_window_end)
"#
END
```

## Algorithm: DetectCrossing (Runtime Logic)

```
ALGORITHM: DetectCrossing
INPUT:
    prev_value: f64
    current_value: f64
    threshold: f64
    condition: Condition
OUTPUT: Option<CrossingDirection>

DESCRIPTION: Describes the crossing detection logic that the SQL implements

BEGIN
    RETURN MATCH condition WITH
        | Condition::LessThan =>
            // Objective: metric < threshold (e.g., CO2 < 800)
            // Rising = entering violation (was ok, now bad)
            // Falling = leaving violation (was bad, now ok)
            IF prev_value < threshold AND current_value >= threshold THEN
                Some(CrossingDirection::Rising)
            ELSE IF prev_value >= threshold AND current_value < threshold THEN
                Some(CrossingDirection::Falling)
            ELSE
                None
            END IF

        | Condition::GreaterThan =>
            // Objective: metric > threshold (e.g., temp > 18)
            // Rising = entering violation (was ok, now bad)
            IF prev_value > threshold AND current_value <= threshold THEN
                Some(CrossingDirection::Rising)
            ELSE IF prev_value <= threshold AND current_value > threshold THEN
                Some(CrossingDirection::Falling)
            ELSE
                None
            END IF

        | Condition::LessOrEqual =>
            IF prev_value <= threshold AND current_value > threshold THEN
                Some(CrossingDirection::Rising)
            ELSE IF prev_value > threshold AND current_value <= threshold THEN
                Some(CrossingDirection::Falling)
            ELSE
                None
            END IF

        | Condition::GreaterOrEqual =>
            IF prev_value >= threshold AND current_value < threshold THEN
                Some(CrossingDirection::Rising)
            ELSE IF prev_value < threshold AND current_value >= threshold THEN
                Some(CrossingDirection::Falling)
            ELSE
                None
            END IF

        | Condition::Between(min, max) =>
            was_in_range <- prev_value >= min AND prev_value <= max
            now_in_range <- current_value >= min AND current_value <= max

            IF NOT was_in_range AND now_in_range THEN
                Some(CrossingDirection::EnteringRange)
            ELSE IF was_in_range AND current_value < min THEN
                Some(CrossingDirection::ExitingRangeLow)
            ELSE IF was_in_range AND current_value > max THEN
                Some(CrossingDirection::ExitingRangeHigh)
            ELSE
                None
            END IF
END
```

## SQL Example: Indoor Air Quality Threshold Crossings

```sql
-- Generated threshold crossing view
CREATE OR REPLACE VIEW gold.indoor_air_quality_threshold_crossings AS

WITH
-- Objective thresholds (inlined from domain config)
objective_thresholds AS (
    SELECT
        'healthy_co2'::TEXT AS objective_id,
        'air-quality'::TEXT AS stream_id,
        'co2'::TEXT AS metric,
        '<'::TEXT AS condition,
        800::NUMERIC AS threshold,
        NULL::NUMERIC AS threshold_min,
        NULL::NUMERIC AS threshold_max,
        'ppm'::TEXT AS unit,
        NULL::TIME AS time_window_start,
        NULL::TIME AS time_window_end
    UNION ALL
    SELECT
        'healthy_pm25'::TEXT AS objective_id,
        'air-quality'::TEXT AS stream_id,
        'pm25'::TEXT AS metric,
        '<'::TEXT AS condition,
        12::NUMERIC AS threshold,
        NULL::NUMERIC AS threshold_min,
        NULL::NUMERIC AS threshold_max,
        'ug/m3'::TEXT AS unit,
        NULL::TIME AS time_window_start,
        NULL::TIME AS time_window_end
),

-- Air quality observations with LAG
air_quality_observations AS (
    SELECT
        bucket,
        ndp_id AS entity_id,
        'air-quality'::TEXT AS stream_id,
        co2_mean AS co2_value,
        LAG(co2_mean) OVER (PARTITION BY ndp_id ORDER BY bucket) AS co2_prev,
        pm25_mean AS pm25_value,
        LAG(pm25_mean) OVER (PARTITION BY ndp_id ORDER BY bucket) AS pm25_prev
    FROM gold.air_quality_hourly
),

-- Unpivot to metric rows
observation_metrics AS (
    SELECT
        bucket,
        entity_id,
        stream_id,
        metric,
        value,
        prev_value
    FROM air_quality_observations
    CROSS JOIN LATERAL (
        VALUES
            ('co2', co2_value, co2_prev),
            ('pm25', pm25_value, pm25_prev)
    ) AS metrics(metric, value, prev_value)
    WHERE value IS NOT NULL AND prev_value IS NOT NULL
),

-- Detect crossings
crossings AS (
    SELECT
        om.bucket AS event_time,
        om.stream_id,
        om.entity_id,
        ot.objective_id,
        ot.metric,
        ot.threshold,
        ot.condition,
        ot.unit,
        om.value,
        om.prev_value,
        CASE
            WHEN ot.condition = '<' THEN
                CASE
                    WHEN om.prev_value < ot.threshold AND om.value >= ot.threshold THEN 'rising'
                    WHEN om.prev_value >= ot.threshold AND om.value < ot.threshold THEN 'falling'
                END
        END AS direction
    FROM observation_metrics om
    JOIN objective_thresholds ot
        ON om.stream_id = ot.stream_id
        AND om.metric = ot.metric
)

-- Final output
SELECT
    gen_random_uuid() AS event_id,
    event_time,
    stream_id,
    entity_id,
    'threshold_crossing'::TEXT AS event_type,
    jsonb_build_object(
        'metric', metric,
        'threshold', threshold,
        'direction', direction,
        'value', value,
        'previous_value', prev_value,
        'objective_id', objective_id,
        'condition', condition,
        'unit', unit
    ) AS details
FROM crossings
WHERE direction IS NOT NULL;
```

## Complexity Analysis

| Operation | Time Complexity | Space Complexity |
|-----------|-----------------|------------------|
| LAG computation | O(n log n) | O(n) |
| Unpivot (CROSS JOIN LATERAL) | O(n * m) | O(n * m) |
| Objective join | O(n * o) | O(n * o) |
| Direction CASE | O(1) per row | O(1) |
| Total view query | O(n * m * o) | O(n * m) |

Where: n = observations, m = metrics, o = objectives

## Invariants

1. **Both Values Required**: Crossings only detected when prev AND current are non-NULL
2. **Per-Entity LAG**: LAG partitioned by entity_id
3. **Time-Ordered**: LAG ordered by bucket timestamp
4. **Direction Non-NULL**: Only actual crossings have direction (filter WHERE direction IS NOT NULL)
5. **Time Window Applied**: If objective has time_window, only detect during that window

## Monitoring Queries

### Crossing Frequency per Objective

```sql
SELECT
    DATE_TRUNC('day', event_time) AS day,
    details->>'objective_id' AS objective_id,
    COUNT(*) AS crossing_count
FROM gold.events_unified
WHERE event_type = 'threshold_crossing'
GROUP BY 1, 2
ORDER BY 1 DESC, 3 DESC;
```

### Oscillation Detection (Rapid Crossings)

```sql
WITH crossing_gaps AS (
    SELECT
        event_time,
        details->>'objective_id' AS objective_id,
        event_time - LAG(event_time) OVER (
            PARTITION BY details->>'objective_id'
            ORDER BY event_time
        ) AS gap_since_previous
    FROM gold.events_unified
    WHERE event_type = 'threshold_crossing'
)
SELECT
    objective_id,
    DATE_TRUNC('day', event_time) AS day,
    COUNT(*) FILTER (WHERE gap_since_previous < INTERVAL '1 hour') AS oscillations
FROM crossing_gaps
WHERE gap_since_previous IS NOT NULL
GROUP BY 1, 2
HAVING COUNT(*) FILTER (WHERE gap_since_previous < INTERVAL '1 hour') > 5;
```
