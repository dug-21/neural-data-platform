# ALGO-data-dictionary-population: Gold Layer Metadata Population

> **Algorithm ID:** D02
> **Feature:** v11-010 (Gold Layer Data Dictionary)
> **Phase:** D (Validation + Dashboard)
> **Created:** 2026-02-04

---

## Purpose

Populate the data dictionary with Gold layer metadata during DDL generation and deployment. This algorithm ensures that all Gold tables, columns, and their lineage are documented in `data_dictionary.gold_tables` and `data_dictionary.gold_columns`, enabling MCP tools, dashboards, and V1.2 pattern detection to discover available data.

---

## Algorithm: PopulateGoldMetadata

```
ALGORITHM: PopulateGoldMetadata
INPUT:
    stream_config: StreamConfig
    granularity: String
    gold_etl: GoldEtlConfig
    action: Action
OUTPUT: Result<String, GeneratorError>
REQUIRES:
    - data_dictionary.gold_tables exists
    - data_dictionary.gold_columns exists

BEGIN
    // 1. Derive table name from stream and granularity
    table_name <- DeriveGoldTableName(stream_config.stream_id, granularity)
    view_short_name <- table_name.split(".").last()

    // 2. Determine source Silver table
    source_silver_table <- format!(
        "silver.{}",
        stream_config.stream_id.replace("-", "_") + "_observations"
    )

    // 3. Generate table metadata insert
    table_sql <- GenerateTableMetadataInsert(
        view_short_name,
        source_silver_table,
        stream_config.stream_id,
        granularity,
        gold_etl
    )

    // 4. Generate column metadata inserts
    column_sql <- GenerateColumnMetadataInserts(
        view_short_name,
        gold_etl
    )

    // 5. Combine into atomic transaction block
    combined_sql <- format!(r#"
-- Data Dictionary Population for {table_name}
-- Generated by ndp-gold-ddl

BEGIN;

{table_insert}

{column_inserts}

COMMIT;
"#,
        table_name = table_name,
        table_insert = table_sql,
        column_inserts = column_sql
    )

    RETURN Ok(combined_sql)
END
```

## Algorithm: GenerateTableMetadataInsert

```
ALGORITHM: GenerateTableMetadataInsert
INPUT:
    view_short_name: String
    source_silver_table: String
    stream_id: String
    granularity: String
    gold_etl: GoldEtlConfig
OUTPUT: String

BEGIN
    // Extract refresh interval from config
    refresh_interval <- gold_etl.refresh_policy
        .map(|rp| rp.schedule_interval)
        .unwrap_or("15 minutes")

    sql <- format!(r#"
-- Upsert gold_tables metadata
INSERT INTO data_dictionary.gold_tables (
    table_name,
    table_schema,
    object_type,
    source_silver_table,
    source_stream_id,
    bucket_interval,
    refresh_interval,
    description,
    updated_at
)
VALUES (
    '{view_short_name}',
    'gold',
    'continuous_aggregate',
    '{source_silver_table}',
    '{stream_id}',
    INTERVAL '{granularity}',
    INTERVAL '{refresh_interval}',
    '{description}',
    NOW()
)
ON CONFLICT (table_name) DO UPDATE SET
    source_silver_table = EXCLUDED.source_silver_table,
    source_stream_id = EXCLUDED.source_stream_id,
    bucket_interval = EXCLUDED.bucket_interval,
    refresh_interval = EXCLUDED.refresh_interval,
    description = EXCLUDED.description,
    updated_at = NOW();
"#,
        view_short_name = view_short_name,
        source_silver_table = source_silver_table,
        stream_id = stream_id,
        granularity = granularity,
        refresh_interval = refresh_interval,
        description = EscapeSqlString(gold_etl.description.unwrap_or(""))
    )

    RETURN sql
END
```

## Algorithm: GenerateColumnMetadataInserts

```
ALGORITHM: GenerateColumnMetadataInserts
INPUT:
    table_name: String
    gold_etl: GoldEtlConfig
OUTPUT: String

BEGIN
    inserts <- Vec::new()
    aggregates <- gold_etl.aggregates.unwrap()

    // 1. Bucket column (dimension)
    inserts.push(GenerateSingleColumnInsert(
        table_name,
        "bucket",
        "timestamptz",
        "dimension",
        None,      // source_field
        "time_bucket(...)",
        None,      // metric_type
        None,      // lag_hours
        None,      // window_hours
        None       // unit
    ))

    // 2. Entity column (identity)
    inserts.push(GenerateSingleColumnInsert(
        table_name,
        aggregates.entity_column,
        "text",
        "identity",
        None,
        aggregates.entity_column,
        None, None, None, None
    ))

    // 3. Aggregate columns
    FOR EACH (field_name, field_config) IN aggregates.fields DO
        unit <- field_config.unit.clone()

        FOR EACH metric IN field_config.metrics DO
            column_name <- format!("{}_{}", field_name, metric)
            source_expr <- GenerateSourceExpression(field_name, metric)

            inserts.push(GenerateSingleColumnInsert(
                table_name,
                column_name,
                DetermineDataType(metric),
                "aggregate",
                Some(field_name),
                source_expr,
                Some(metric),
                None, None,
                unit.clone()
            ))
        END FOR
    END FOR

    // 4. Sample count column (aggregate)
    inserts.push(GenerateSingleColumnInsert(
        table_name,
        "sample_count",
        "bigint",
        "aggregate",
        None,
        "COUNT(*)",
        Some("count"),
        None, None, None
    ))

    // 5. Feature columns (if configured)
    IF gold_etl.features IS Some(features) THEN
        feature_inserts <- GenerateFeatureColumnInserts(table_name, features)
        inserts.extend(feature_inserts)
    END IF

    // Combine with delete-then-insert pattern for full sync
    combined_sql <- format!(r#"
-- Clear existing column metadata for this table
DELETE FROM data_dictionary.gold_columns WHERE table_name = '{table_name}';

-- Insert column metadata
{inserts}
"#,
        table_name = table_name,
        inserts = inserts.join("\n")
    )

    RETURN combined_sql
END
```

## Algorithm: GenerateSingleColumnInsert

```
ALGORITHM: GenerateSingleColumnInsert
INPUT:
    table_name: String
    column_name: String
    data_type: String
    feature_type: String
    source_field: Option<String>
    source_expression: String
    metric_type: Option<String>
    lag_hours: Option<i32>
    window_hours: Option<i32>
    unit: Option<String>
OUTPUT: String

BEGIN
    sql <- format!(r#"
INSERT INTO data_dictionary.gold_columns (
    table_name,
    column_name,
    data_type,
    feature_type,
    source_field,
    source_expression,
    metric_type,
    lag_hours,
    window_hours,
    unit
)
VALUES (
    '{table_name}',
    '{column_name}',
    '{data_type}',
    '{feature_type}',
    {source_field_sql},
    '{source_expression}',
    {metric_type_sql},
    {lag_hours_sql},
    {window_hours_sql},
    {unit_sql}
);"#,
        table_name = table_name,
        column_name = column_name,
        data_type = data_type,
        feature_type = feature_type,
        source_field_sql = source_field.map(|s| format!("'{}'", s)).unwrap_or("NULL"),
        source_expression = source_expression,
        metric_type_sql = metric_type.map(|s| format!("'{}'", s)).unwrap_or("NULL"),
        lag_hours_sql = lag_hours.map(|i| i.to_string()).unwrap_or("NULL"),
        window_hours_sql = window_hours.map(|i| i.to_string()).unwrap_or("NULL"),
        unit_sql = unit.map(|s| format!("'{}'", s)).unwrap_or("NULL")
    )

    RETURN sql
END
```

## Algorithm: GenerateFeatureColumnInserts

```
ALGORITHM: GenerateFeatureColumnInserts
INPUT:
    table_name: String
    features: FeaturesConfig
OUTPUT: Vec<String>

BEGIN
    inserts <- Vec::new()

    // 1. Lag feature columns
    IF features.lag IS Some(lag) AND lag.enabled THEN
        FOR EACH field IN lag.fields DO
            FOR EACH lag_hours IN lag.lags_hours DO
                column_name <- format!("{}_mean_lag_{}h", field, lag_hours)
                source_expr <- format!("LAG({}_mean, {}) OVER ...", field, lag_hours)

                inserts.push(GenerateSingleColumnInsert(
                    table_name,
                    column_name,
                    "double precision",
                    "lag",
                    Some(field),
                    source_expr,
                    None,
                    Some(lag_hours),
                    None,
                    None
                ))
            END FOR
        END FOR
    END IF

    // 2. Rolling feature columns
    IF features.rolling IS Some(rolling) AND rolling.enabled THEN
        FOR EACH field IN rolling.fields DO
            FOR EACH window IN rolling.windows DO
                window_hours <- ParseWindowToHours(window)

                FOR EACH stat IN rolling.stats DO
                    column_name <- format!("{}_roll_{}h_{}", field, window_hours, stat)
                    source_expr <- format!("{}({}) OVER (ROWS {} PRECEDING)", stat.to_upper(), field, window_hours)

                    inserts.push(GenerateSingleColumnInsert(
                        table_name,
                        column_name,
                        "double precision",
                        "rolling",
                        Some(field),
                        source_expr,
                        Some(stat),
                        None,
                        Some(window_hours),
                        None
                    ))
                END FOR
            END FOR
        END FOR
    END IF

    // 3. Trend feature columns
    IF features.trend IS Some(trend) AND trend.enabled THEN
        window_hours <- ParseWindowToHours(trend.window)

        FOR EACH field IN trend.fields DO
            column_name <- format!("{}_trend_{}h", field, window_hours)
            source_expr <- format!("(current - lag) / NULLIF(lag, 0) over {} hours", window_hours)

            inserts.push(GenerateSingleColumnInsert(
                table_name,
                column_name,
                "double precision",
                "trend",
                Some(field),
                source_expr,
                None,
                None,
                Some(window_hours),
                None
            ))
        END FOR
    END IF

    RETURN inserts
END
```

## Algorithm: DetectFeatureType

```
ALGORITHM: DetectFeatureType
INPUT: column_name: String
OUTPUT: String
DESCRIPTION: Auto-detect feature type from column naming pattern

BEGIN
    RETURN MATCH column_name WITH
        // Aggregate patterns
        | name IF name.ends_with("_mean") => "aggregate"
        | name IF name.ends_with("_std") => "aggregate"
        | name IF name.ends_with("_min") => "aggregate"
        | name IF name.ends_with("_max") => "aggregate"
        | name IF name.ends_with("_p95") => "aggregate"
        | name IF name.ends_with("_p99") => "aggregate"
        | name IF name.ends_with("_count") => "aggregate"
        | name IF name.ends_with("_sum") => "aggregate"
        | "sample_count" => "aggregate"

        // Lag patterns
        | name IF name.contains("_lag_") => "lag"

        // Rolling patterns
        | name IF name.contains("_roll_") => "rolling"

        // Trend patterns
        | name IF name.contains("_trend_") => "trend"

        // Special columns
        | "bucket" => "dimension"
        | "ndp_id" => "identity"
        | name IF name.ends_with("_ndp_id") => "identity"

        // Default
        | _ => "raw"
END
```

## Algorithm: VerifyDataDictionaryCompleteness

```
ALGORITHM: VerifyDataDictionaryCompleteness
INPUT: db_executor: DatabaseExecutor
OUTPUT: Result<CompletenessReport, QueryError>

BEGIN
    report <- CompletenessReport::new()

    // 1. Check all Gold tables are documented
    missing_tables <- db_executor.query(r#"
        SELECT 'MISSING: ' || v.view_name as issue
        FROM timescaledb_information.continuous_aggregates v
        LEFT JOIN data_dictionary.gold_tables gt
            ON v.view_name = gt.table_name
        WHERE v.view_schema = 'gold'
          AND gt.table_name IS NULL
    "#)?

    report.missing_tables <- missing_tables.rows

    // 2. Check all streams are classified
    unclassified_streams <- db_executor.query(r#"
        SELECT 'UNCLASSIFIED: ' || s.stream_id as issue
        FROM data_dictionary.streams s
        LEFT JOIN data_dictionary.stream_classification sc
            ON s.stream_id = sc.stream_id
        WHERE sc.stream_id IS NULL
    "#)?

    report.unclassified_streams <- unclassified_streams.rows

    // 3. Check Gold columns match actual columns
    missing_columns <- db_executor.query(r#"
        SELECT 'MISSING: ' || c.table_name || '.' || c.column_name as issue
        FROM information_schema.columns c
        JOIN data_dictionary.gold_tables gt ON c.table_name = gt.table_name
        LEFT JOIN data_dictionary.gold_columns gc
            ON c.table_name = gc.table_name
            AND c.column_name = gc.column_name
        WHERE c.table_schema = 'gold'
          AND gc.column_name IS NULL
    "#)?

    report.missing_columns <- missing_columns.rows

    // 4. Check all columns have feature type
    missing_feature_type <- db_executor.query(r#"
        SELECT 'NO_TYPE: ' || table_name || '.' || column_name as issue
        FROM data_dictionary.gold_columns
        WHERE feature_type IS NULL OR feature_type = ''
    "#)?

    report.missing_feature_types <- missing_feature_type.rows

    // 5. Calculate completeness metrics
    report.table_coverage <- CalculateCoverage(report.missing_tables)
    report.column_coverage <- CalculateCoverage(report.missing_columns)
    report.classification_coverage <- CalculateCoverage(report.unclassified_streams)
    report.feature_type_coverage <- CalculateCoverage(report.missing_feature_types)

    RETURN Ok(report)
END
```

## SQL Templates

### Query: List All Gold Layer Objects

```sql
SELECT
    gt.table_name,
    gt.object_type,
    gt.source_stream_id,
    gt.bucket_interval,
    gt.refresh_interval,
    COUNT(gc.column_name) as column_count
FROM data_dictionary.gold_tables gt
LEFT JOIN data_dictionary.gold_columns gc ON gt.table_name = gc.table_name
GROUP BY gt.table_name, gt.object_type, gt.source_stream_id,
         gt.bucket_interval, gt.refresh_interval
ORDER BY gt.table_name;
```

### Query: List Columns by Feature Type

```sql
SELECT
    gc.table_name,
    gc.column_name,
    gc.feature_type,
    gc.source_field,
    gc.metric_type,
    gc.lag_hours,
    gc.window_hours,
    gc.unit
FROM data_dictionary.gold_columns gc
WHERE gc.feature_type = :feature_type  -- 'aggregate', 'lag', 'rolling', 'trend'
ORDER BY gc.table_name, gc.column_name;
```

### Query: Trace Column Lineage

```sql
SELECT
    gc.table_name as gold_table,
    gc.column_name as gold_column,
    gc.feature_type,
    gc.source_expression,
    gt.source_silver_table,
    gc.source_field as silver_field
FROM data_dictionary.gold_columns gc
JOIN data_dictionary.gold_tables gt ON gc.table_name = gt.table_name
WHERE gc.source_field IS NOT NULL
ORDER BY gc.table_name, gc.column_name;
```

## Complexity Analysis

| Operation | Time Complexity | Space Complexity |
|-----------|-----------------|------------------|
| Generate table insert | O(1) | O(1) |
| Generate column inserts | O(f * m) | O(f * m) |
| Generate feature inserts | O(f * l + f * w) | O(f * l + f * w) |
| Verify completeness | O(t + c) | O(issues) |

Where: f = fields, m = metrics, l = lag depths, w = windows, t = tables, c = columns

## Invariants

1. **Atomic Population**: All metadata in single transaction
2. **Full Sync**: Column metadata is delete-then-insert
3. **Upsert Tables**: Table metadata uses ON CONFLICT for updates
4. **Feature Type Assigned**: Every column has a feature_type
5. **Source Expression**: Every computed column has source_expression

## Integration with deploy.sh

```bash
# In deploy.sh handle_gold_table()
handle_gold_table() {
    local declaration="$1"
    local stream_id=$(echo "$declaration" | jq -r '.stream_id')
    local action=$(echo "$declaration" | jq -r '.action // "sync"')

    log "Gold Table: $stream_id (action=$action)"

    # Generate DDL (includes metadata inserts)
    local ddl=$(ndp-gold-ddl generate --stream "$stream_id" --mode "$action" --include-metadata 2>&1)

    # Execute DDL (DDL + metadata in same transaction)
    echo "$ddl" | dcx timescaledb psql -U postgres -d ndp
}
```
